# מבנה קובץ תוכן לשיעור (content/lesson-NN.js)

כל שיעור הוא קובץ JS רגיל (לא מודול) שמוסיף אובייקט אחד ל־`window.LESSONS`.
הדף מציג כל שיעור כ״מצגת בגלילה״: שקופית פתיחה (נבנית אוטומטית מ־goals/mat),
אחריה שקופית לכל שלב במהלך ה־45 דקות **לפי הסדר**, ובסוף שקופית שיעורי בית
(נבנית אוטומטית מ־hw). כל שקופית מחולקת ל:
- **מסך** — מה שמוקרן לכיתה (שאלה גדולה, הוראות, תחילות משפטים).
- **למורה** — מה להגיד ולהסביר, שאלות מנחות, תפיסות שגויות. מוסתר ב״מצב הקרנה״.

```js
(window.LESSONS = window.LESSONS || []).push({
  n: 1, c: 3, t: 'sci',            // 'sci' מדעים | 'lit' אוריינות
  q: 'מדוע השמיים כחולים?',
  p: '11–16',                      // עמודים בחוברת
  cover: 'sky',                    // מזהה סצנת פתיחה מונפשת (ראו רשימה)
  goals: [['מטרה מדעית', '...'], ['מטרת חשיבה וכתיבה', '...']],  // מהבריף, מילה במילה
  mat: ['...', '...'],             // חומרים, מהבריף
  check: '...',                    // ״בדיקת הבנה״/״תוצר״/״הערכת התוצר״ מהבריף (או null)
  checkLabel: 'בדיקת הבנה',
  deep: '...',                     // ״העמקה לבחירה״ מהבריף (או null)
  hw: '...',                       // שיעורי הבית מהבריף, מילה במילה
  slides: [
    {
      from: 0, to: 5,              // דקות — בדיוק כמו בבריף. שקופית אחת לכל שלב במהלך.
      kind: 'open',                // open | predict | explore | explain | write | reflect | close
      mode: 'solo',                // solo (לבד) | pair (זוגות) | group (קבוצות) | class (מליאה)
      title: 'כותרת קצרה למסך',     // 2–5 מילים
      lead: 'השאלה או ההוראה הגדולה שמוקרנת לכיתה',
      steps: ['הוראה 1', 'הוראה 2'],          // אופציונלי: הוראות פעילות ממוספרות, קצרות, בגוף שני רבים
      starters: ['אור השמש…', 'ולכן…'],       // אופציונלי: תחילות משפטים (לא פסקה שלמה!)
      gate: {                                  // אופציונלי: ״קודם ניחושים, אחר כך חשיפה״
        prompt: 'מי יכולה להציע הסבר? כל אחת.ד רושמת ניחוש במחברת.',
        options: ['האוויר כחול', 'השמיים משקפים את הים', 'משהו קורה לאור באוויר'], // אופציונלי: המורה סופרת הצבעות
        reveal: 'מה נחשף אחרי הניחושים (שאלת המשך / רמז / הכיוון הבא — לא בהכרח התשובה)'
      },
      sim: 'sky',                  // אופציונלי: מזהה הדמיה אינטראקטיבית (ראו רשימה)
      simNote: 'איך להשתמש בהדמיה בשלב הזה (משפט אחד)',
      teacher: {
        say: ['מה המורה אומרת/מסבירה — 2–5 משפטים קצרים שאפשר לומר בכיתה כלשונם'],
        ask: ['שאלות מנחות / שאלות המשך'],
        watch: ['תפיסה שגויה או דבר לשים לב אליו'],
        tip: 'טיפ ניהולי אחד קצר (אופציונלי)'
      }
    }
  ]
});
```

## כללים
- עברית, בפנייה כפולה מגדרית בסגנון הבריף: ״כותבות.ים״, ״תלמידות.ים״, ״כל אחת.ד״.
- **בתוך מחרוזות JS להשתמש רק בגרשיים עבריים ״ ״ ובגרש ׳** — לא ב־" או ' רגילים, כדי לא לשבור את המחרוזת. מחרוזות עטופות ב־' רגיל.
- דיוק מדעי מלא. אין ״הסבר שגוי מפשט״. כשמשהו הוא אנלוגיה — לומר זאת.
- לא מציגים פסקה שלמה להעתקה בשלבי כתיבה; רק מושגים, שאלות ותחילות משפטים.
- כל שאלת פתיחה / ניבוי — עם `gate`: קודם ניחושים, ורק אחר כך חשיפה. ההסבר המדעי מגיע **אחרי** ההתנסות.
- טון חם, ישיר, סקרן; לא מתיילד. משפטי מסך קצרים (המסך מוקרן).
- נתונים בדויים מסומנים ״נתונים בדויים לתרגול״.
- אין משימות שמחייבות טלפון אישי. רכיבים דיגיטליים — במסך המשותף בלבד.
